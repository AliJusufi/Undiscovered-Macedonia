function localAnswer(text) {
  var q = text.toLowerCase();
  var best = null, bestScore = 0;
  KB.forEach(function (entry) {
    var score = 0;
    entry.k.forEach(function (kw, i) {
      if (q.indexOf(kw) !== -1) {
        // first keyword = canonical name, weighted 4x
        score += (i === 0 ? kw.length * 4 : kw.length);
      }
    });
    if (score > bestScore) { bestScore = score; best = entry; }
  });
  return best ? best.a
              : "I can only answer from a small built-in guide right now ...";
}

function respond(text) {
  if (ASSISTANT_CONFIG.apiUrl) {
    return fetch(ASSISTANT_CONFIG.apiUrl, {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({ messages: messages })
    })
      .then(function (r) {
        if (!r.ok) throw new Error("proxy " + r.status);
        return r.json();
      })
      .then(function (data) { return data.reply || localAnswer(text); })
      .catch(function () { return localAnswer(text); });  // proxy down -> offline guide
  }
  // no proxy configured: offline answer after a short "thinking" delay
  return new Promise(function (resolve) {
    setTimeout(function () { resolve(localAnswer(text)); }, 400);
  });
}

function decide(p, kind) {
  seen[p.id] = kind;
  LS.set(K_SEEN, seen);
  queue = queue.filter(function (q) { return q.id !== p.id; });

  var matched = false;
  if (kind === "like") {
    var sh = sharedInterests(p).length;
    matched = sh >= 2 || Math.random() < 0.35;
    if (matched && matches.indexOf(p.id) === -1) {
      matches.push(p.id);
      LS.set(K_MATCHES, matches);
      chats[p.id] = [{ from: "them", text: openingLine(p) }];
      LS.set(K_CHATS, chats);
    }
  }
  setTimeout(function () { renderDeck(); if (matched) showMatch(p); }, 260);
}

MATCHING-BACKEND.md (Appendix D) 
create table profiles (
  id uuid primary key references auth.users on delete cascade,
  name text not null, country text, age int,
  pace text, from_date date, to_date date,
  interests text[] not null default '{}', bio text
);
create table swipes (
  swiper uuid references profiles(id), target uuid references profiles(id),
  liked boolean not null, primary key (swiper, target)
);
create table matches (
  a uuid references profiles(id), b uuid references profiles(id),
  primary key (a, b), check (a < b)
);

-- a mutual "like" becomes a match automatically
create function on_swipe() returns trigger as $$
begin
  if new.liked and exists (
    select 1 from swipes
    where swiper = new.target and target = new.swiper and liked
  ) then
    insert into matches (a, b)
    values (least(new.swiper, new.target), greatest(new.swiper, new.target))
    on conflict do nothing;
  end if;
  return new;
end; $$ language plpgsql;
